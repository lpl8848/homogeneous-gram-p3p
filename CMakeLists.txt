cmake_minimum_required(VERSION 3.16)
project(homogeneous_gram_p3p VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)

add_library(gram_p3p STATIC third_party/poselib/PoseLib/misc/univariate.cc)
add_library(gram_p3p::gram_p3p ALIAS gram_p3p)
target_include_directories(gram_p3p PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/third_party/poselib>)
target_link_libraries(gram_p3p PUBLIC Eigen3::Eigen)

add_executable(p3p_example examples/minimal.cpp)
add_executable(p3p_simulation simulations/simulation.cpp)
add_executable(p3p_tests tests/test_solver.cpp)
foreach(target gram_p3p p3p_example p3p_simulation p3p_tests)
    if(NOT target STREQUAL "gram_p3p")
        target_link_libraries(${target} PRIVATE gram_p3p::gram_p3p)
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE /O2 /fp:precise)
    else()
        target_compile_options(${target} PRIVATE $<$<CONFIG:Release>:-O3>)
    endif()
endforeach()
enable_testing()
add_test(NAME solver_geometry COMMAND p3p_tests)
